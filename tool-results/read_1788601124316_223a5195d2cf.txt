   104→const ID_DATE = new Intl.DateTimeFormat('id-ID', {
   105→  day: '2-digit',
   106→  month: 'short',
   107→  year: 'numeric',
   108→})
   109→
   110→function fmtDate(iso: string | null): string {
   111→  if (!iso) return '—'
   112→  const d = new Date(iso)
   113→  if (Number.isNaN(d.getTime())) return '—'
   114→  return ID_DATE.format(d)
   115→}
   116→
   117→function fmtViews(n: number): string {
   118→  return n.toLocaleString('id-ID')
   119→}

   121→/** Map kategori color key → Tailwind badge style. */
   122→function categoryBadgeClass(color: string | null | undefined): string {
   123→  switch (color) {
   124→    case 'amber':
   125→      return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/40 dark:text-amber-200 dark:border-amber-800'
   126→    case 'red':
   127→      return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/40 dark:text-red-200 dark:border-red-800'
   128→    case 'emerald':
   129→      return 'bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/40 dark:text-emerald-200 dark:border-emerald-800'
   130→    case 'slate':
   131→      return 'bg-slate-200 text-slate-800 border-slate-300 dark:bg-slate-700/60 dark:text-slate-100 dark:border-slate-600'
   132→    default:

   154→function FilterBar({
   155→  q,
   156→  categoryFilter,
   157→  statusFilter,
   158→  sort,
   159→  categories,
   160→}: {
   161→  q: string
   162→  categoryFilter: string
   163→  statusFilter: string
   164→  sort: string
   165→  categories: { id: string; name: string; slug: string; color: string | null }[]
   166→}) {
   167→  const router = useRouter()
   168→  const pathname = usePathname()
   169→  const sp = useSearchParams()
   170→
   171→  // Local input state untuk search, push ke URL saat submit / debounce.
   172→  const [searchInput, setSearchInput] = useState(q)
   173→
   174→  function pushParams(next: Record<string, string>) {
   175→    const params = new URLSearchParams(sp.toString())
   176→    // Reset ke page 1 setiap kali filter berubah
   177→    if (
   178→      'q' in next ||
   179→      'category' in next ||
   180→      'status' in next ||
   181→      'sort' in next
   182→    ) {
   183→      params.delete('page')
   184→    }
   185→    for (const [k, v] of Object.entries(next)) {
   186→      if (v === '' || v === 'all') params.delete(k)
   187→      else params.set(k, v)
   188→    }
   189→    router.push(`${pathname}?${params.toString()}`, { scroll: false })
   190→  }
   191→
   192→  return (
   193→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   194→      <div className="relative flex-1">
   195→        <Search className="pointer-events-none absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   196→        <Input
   197→          type="search"
   198→          value={searchInput}
   199→          onChange={(e) => setSearchInput(e.target.value)}
   200→          onKeyDown={(e) => {
   201→            if (e.key === 'Enter') pushParams({ q: searchInput })
   202→          }}
   203→          onBlur={() => pushParams({ q: searchInput })}
   204→          placeholder="Cari judul artikel..."
   205→          className="pl-8"
   206→          aria-label="Cari artikel"
   207→        />
   208→      </div>
   209→      <div className="flex flex-wrap gap-2">
   210→        <Select
   211→          value={categoryFilter || 'all'}
   212→          onValueChange={(v) => pushParams({ category: v })}
   213→        >
   214→          <SelectTrigger className="w-[160px]" aria-label="Filter kategori">
   215→            <SelectValue placeholder="Kategori" />
   216→          </SelectTrigger>
   217→          <SelectContent>
   218→            <SelectItem value="all">Semua Kategori</SelectItem>
   219→            {categories.map((c) => (
   220→              <SelectItem key={c.id} value={c.slug}>
   221→                {c.name}
   222→              </SelectItem>
   223→            ))}
   224→          </SelectContent>
   225→        </Select>
   226→        <Select
   227→          value={statusFilter || 'all'}
   228→          onValueChange={(v) => pushParams({ status: v })}
   229→        >
   230→          <SelectTrigger className="w-[140px]" aria-label="Filter status">
   231→            <SelectValue placeholder="Status" />
   232→          </SelectTrigger>
   233→          <SelectContent>
   234→            <SelectItem value="all">Semua Status</SelectItem>
   235→            <SelectItem value="DRAFT">Draft</SelectItem>
   236→            <SelectItem value="PUBLISHED">Published</SelectItem>
   237→            <SelectItem value="ARCHIVED">Archived</SelectItem>
   238→          </SelectContent>
   239→        </Select>
   240→        <Select value={sort} onValueChange={(v) => pushParams({ sort: v })}>
   241→          <SelectTrigger className="w-[160px]" aria-label="Urutkan">
   242→            <SelectValue placeholder="Urutkan" />
   243→          </SelectTrigger>
   244→          <SelectContent>
   245→            <SelectItem value="newest">Terbaru</SelectItem>
   246→            <SelectItem value="oldest">Terlama</SelectItem>
   247→            <SelectItem value="popular">Populer (views)</SelectItem>
   248→            <SelectItem value="title-asc">Judul A-Z</SelectItem>
   249→          </SelectContent>
   250→        </Select>
   251→        <Button asChild className="bg-gradient-to-r from-amber-500 to-orange-600 text-white hover:from-amber-600 hover:to-orange-700">
   252→          <Link href="/admin/articles/new">
   253→            <Plus className="h-4 w-4" />
   254→            <span className="hidden sm:inline">Artikel Baru</span>
   255→          </Link>
   256→        </Button>
   257→      </div>
   258→    </div>
   259→  )
   260→}

   266→function RowActions({ row }: { row: ArticleListItem }) {
   267→  const [confirmDelete, setConfirmDelete] = useState(false)
   268→  const [pending, startTransition] = useTransition()
   269→  const router = useRouter()
   270→
   271→  function toggleFeatured() {
   272→    startTransition(async () => {
   273→      try {
   274→        const res = await fetch(`/api/admin/articles/${row.id}/toggle-featured`, {
   275→          method: 'POST',
   276→        })
   277→        if (!res.ok) {
   278→          const data = await res.json().catch(() => ({}))
   279→          throw new Error(data.message || 'Gagal toggle featured')
   280→        }
   281→        const data = (await res.json()) as { ok: boolean; isFeatured?: boolean; warning?: string }
   282→        if (data.warning) {
   283→          toast.info('Featured diaktifkan', { description: data.warning })
   284→        } else {
   285→          toast.success(
   286→            data.isFeatured ? 'Ditandai sebagai featured' : 'Featured dimatikan',
   287→          )
   288→        }
   289→        router.refresh()
   290→      } catch (e) {
   291→        toast.error('Gagal mengubah featured', {
   292→          description: e instanceof Error ? e.message : undefined,
   293→        })
   294→      }
   295→    })
   296→  }
   297→
   298→  function archive() {
   299→    startTransition(async () => {
   300→      try {
   301→        const res = await fetch(`/api/admin/articles/${row.id}`, {
   302→          method: 'PUT',
   303→          headers: { 'Content-Type': 'application/json' },
   304→          body: JSON.stringify({ status: 'ARCHIVED' }),
   305→        })
   306→        if (!res.ok) throw new Error('Gagal mengarsipkan')
   307→        toast.success('Artikel diarsipkan')
   308→        router.refresh()
   309→      } catch (e) {
   310→        toast.error('Gagal mengarsipkan', {
   311→          description: e instanceof Error ? e.message : undefined,
   312→        })
   313→      }
   314→    })
   315→  }
   316→
   317→  function doDelete() {
   318→    startTransition(async () => {
   319→      try {
   320→        const res = await fetch(`/api/admin/articles/${row.id}`, { method: 'DELETE' })
   321→        if (!res.ok) throw new Error('Gagal menghapus')
   322→        toast.success('Artikel dihapus')
   323→        setConfirmDelete(false)
   324→        router.refresh()
   325→      } catch (e) {
   326→        toast.error('Gagal menghapus', {
   327→          description: e instanceof Error ? e.message : undefined,
   328→        })
   329→      }
   330→    })
   331→  }
   332→
   333→  return (
   334→    <>
   335→      <DropdownMenu>
   336→        <DropdownMenuTrigger asChild>
   337→          <Button
   338→            variant="ghost"
   339→            size="icon"
   340→            className="h-8 w-8"
   341→            aria-label="Aksi"
   342→            disabled={pending}
   343→          >
   344→            <MoreHorizontal className="h-4 w-4" />
   345→          </Button>
   346→        </DropdownMenuTrigger>
   347→        <DropdownMenuContent align="end">
   348→          <DropdownMenuItem asChild>
   349→            <Link href={`/admin/articles/${row.id}/edit`}>
   350→              <Edit3 className="h-4 w-4" />
   351→              Edit
   352→            </Link>
   353→          </DropdownMenuItem>
   354→          <DropdownMenuItem asChild>
   355→            <Link href={`/admin/articles/${row.id}/preview`} target="_blank">
   356→              <Eye className="h-4 w-4" />
   357→              Preview
   358→            </Link>
   359→          </DropdownMenuItem>
   360→          <DropdownMenuSeparator />
   361→          <DropdownMenuItem onClick={toggleFeatured}>
   362→            {row.isFeatured ? (
   363→              <>
   364→                <StarOff className="h-4 w-4" />
   365→                Lepas Featured
   366→              </>
   367→            ) : (
   368→              <>
   369→                <Star className="h-4 w-4" />
   370→                Jadikan Featured
   371→              </>
   372→            )}
   373→          </DropdownMenuItem>
   374→          <DropdownMenuItem onClick={archive}>
   375→            <Archive className="h-4 w-4" />
   376→            Arsipkan
   377→          </DropdownMenuItem>
   378→          <DropdownMenuSeparator />
   379→          <DropdownMenuItem
   380→            variant="destructive"
   381→            onClick={() => setConfirmDelete(true)}
   382→          >
   383→            <Trash2 className="h-4 w-4" />
   384→            Hapus
   385→          </DropdownMenuItem>
   386→        </DropdownMenuContent>
   387→      </DropdownMenu>
   388→
   389→      <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   390→        <AlertDialogContent>
   391→          <AlertDialogHeader>
   392→            <AlertDialogTitle>Hapus artikel?</AlertDialogTitle>
   393→            <AlertDialogDescription>
   394→              Artikel <strong>&ldquo;{row.title}&rdquo;</strong> akan dihapus permanen
   395→              beserta semua versi, komentar, dan relasi tag. Aksi ini tidak bisa dibatalkan.
   396→            </AlertDialogDescription>
   397→          </AlertDialogHeader>
   398→          <AlertDialogFooter>
   399→            <AlertDialogCancel disabled={pending}>Batal</AlertDialogCancel>
   400→            <AlertDialogAction
   401→              onClick={doDelete}
   402→              disabled={pending}
   403→              className="bg-destructive text-white hover:bg-destructive/90"
   404→            >
   405→              Ya, hapus
   406→            </AlertDialogAction>
   407→          </AlertDialogFooter>
   408→        </AlertDialogContent>
   409→      </AlertDialog>
   410→    </>
   411→  )
   412→}

   416→/* -------------------------------------------------------------------------- */
   417→
   418→function BulkBar({
   419→  selectedIds,
   420→  onClear,
   421→  onDone,
   422→}: {
   423→  selectedIds: Set<string>
   424→  onClear: () => void
   425→  onDone: () => void
   426→}) {
   427→  const [pending, startTransition] = useTransition()
   428→  const count = selectedIds.size
   429→  const [confirm, setConfirm] = useState<null | 'publish' | 'archive' | 'delete'>(null)
   430→  const router = useRouter()
   431→
   432→  if (count === 0) return null
   433→
   434→  function run(action: 'publish' | 'archive' | 'delete') {
   435→    startTransition(async () => {
   436→      try {
   437→        const res = await fetch('/api/admin/articles/bulk', {
   438→          method: 'POST',
   439→          headers: { 'Content-Type': 'application/json' },
   440→          body: JSON.stringify({
   441→            ids: Array.from(selectedIds),
   442→            action,
   443→          }),
   444→        })
   445→        if (!res.ok) {
   446→          const data = await res.json().catch(() => ({}))
   447→          throw new Error(data.message || 'Gagal bulk action')
   448→        }
   449→        const data = (await res.json()) as { ok: boolean; affected?: number }
   450→        const label =
   451→          action === 'publish'
   452→            ? 'dipublikasikan'
   453→            : action === 'archive'
   454→              ? 'diarsipkan'
   455→              : 'dihapus'
   456→        toast.success(`${data.affected ?? count} artikel ${label}`)
   457→        setConfirm(null)
   458→        onClear()
   459→        onDone()
   460→      } catch (e) {
   461→        toast.error('Bulk action gagal', {
   462→          description: e instanceof Error ? e.message : undefined,
   463→        })
   464→      }
   465→    })
   466→  }
   467→
   468→  return (
   469→    <>
   470→      <div className="sticky bottom-3 z-20 mx-auto flex max-w-fit items-center gap-2 rounded-full border bg-card/95 px-3 py-2 shadow-lg backdrop-blur">
   471→        <span className="px-2 text-sm font-medium">
   472→          {count} artikel dipilih
   473→        </span>
   474→        <Button
   475→          size="sm"
   476→          onClick={() => setConfirm('publish')}
   477→          disabled={pending}
   478→          className="bg-emerald-600 text-white hover:bg-emerald-700"
   479→        >
   480→          <CheckCircle2 className="h-4 w-4" />
   481→          Publish
   482→        </Button>
   483→        <Button size="sm" variant="outline" onClick={() => setConfirm('archive')} disabled={pending}>
   484→          <Archive className="h-4 w-4" />
   485→          Arsipkan
   486→        </Button>
   487→        <Button
   488→          size="sm"
   489→          variant="destructive"
   490→          onClick={() => setConfirm('delete')}
   491→          disabled={pending}
   492→        >
   493→          <Trash2 className="h-4 w-4" />
   494→          Hapus
   495→        </Button>
   496→        <Button size="sm" variant="ghost" onClick={onClear} disabled={pending} aria-label="Bersihkan seleksi">
   497→          <X className="h-4 w-4" />
   498→        </Button>
   499→      </div>
   500→
   501→      <AlertDialog open={confirm !== null} onOpenChange={() => setConfirm(null)}>
   502→        <AlertDialogContent>
   503→          <AlertDialogHeader>
   504→            <AlertDialogTitle>
   505→              {confirm === 'delete'
   506→                ? `Hapus ${count} artikel?`
   507→                : confirm === 'archive'
   508→                  ? `Arsipkan ${count} artikel?`
   509→                  : `Publish ${count} artikel?`}
   510→            </AlertDialogTitle>
   511→            <AlertDialogDescription>
   512→              {confirm === 'delete'
   513→                ? 'Artikel yang dipilih akan dihapus permanen beserta versi, komentar, dan relasi.'
   514→                : confirm === 'archive'
   515→                  ? 'Artikel yang dipilih akan disembunyikan dari front-end (status menjadi ARCHIVED).'
   516→                  : 'Artikel yang dipilih akan dipublikasikan. publishedAt di-set ke sekarang bila belum ada.'}
   517→            </AlertDialogDescription>
   518→          </AlertDialogHeader>
   519→          <AlertDialogFooter>
   520→            <AlertDialogCancel disabled={pending}>Batal</AlertDialogCancel>
   521→            <AlertDialogAction
   522→              disabled={pending}
   523→              onClick={() => confirm && run(confirm)}
   524→              className={cn(
   525→                confirm === 'delete' &&
   526→                  'bg-destructive text-white hover:bg-destructive/90',
   527→              )}
   528→            >
   529→              Lanjutkan
   530→            </AlertDialogAction>
   531→          </AlertDialogFooter>
   532→        </AlertDialogContent>
   533→      </AlertDialog>
   534→    </>
   535→  )
   536→}

   541→
   542→function PaginationBar({
   543→  total,
   544→  page,
   545→  pageSize,
   546→}: {
   547→  total: number
   548→  page: number
   549→  pageSize: number
   550→}) {
   551→  const router = useRouter()
   552→  const pathname = usePathname()
   553→  const sp = useSearchParams()
   554→
   555→  const totalPages = Math.max(1, Math.ceil(total / pageSize))
   556→  if (totalPages <= 1) return null
   557→
   558→  function goTo(p: number) {
   559→    const params = new URLSearchParams(sp.toString())
   560→    if (p <= 1) params.delete('page')
   561→    else params.set('page', String(p))
   562→    router.push(`${pathname}?${params.toString()}`, { scroll: false })
   563→  }
   564→
   565→  // Build simple pagination: 1 ... current-1, current, current+1 ... last
   566→  const pages: (number | '...')[] = []
   567→  const add = (p: number) => pages.push(p)
   568→  const addEllipsis = () => pages.push('...')
   569→  if (totalPages <= 7) {
   570→    for (let i = 1; i <= totalPages; i++) add(i)
   571→  } else {
   572→    add(1)
   573→    if (page > 3) addEllipsis()
   574→    for (let i = Math.max(2, page - 1); i <= Math.min(totalPages - 1, page + 1); i++) add(i)
   575→    if (page < totalPages - 2) addEllipsis()
   576→    add(totalPages)
   577→  }
   578→
   579→  return (
   580→    <nav
   581→      aria-label="Pagination"
   582→      className="flex items-center justify-between gap-2 pt-4"
   583→    >
   584→      <p className="text-xs text-muted-foreground">
   585→        Halaman {page} dari {totalPages} &middot; {total.toLocaleString('id-ID')} artikel
   586→      </p>
   587→      <div className="flex items-center gap-1">
   588→        <Button
   589→          variant="outline"
   590→          size="sm"
   591→          disabled={page <= 1}
   592→          onClick={() => goTo(page - 1)}
   593→        >
   594→          Sebelumnya
   595→        </Button>
   596→        {pages.map((p, i) =>
   597→          p === '...' ? (
   598→            <span key={`e${i}`} className="px-2 text-muted-foreground">
   599→              …
   600→            </span>
   601→          ) : (
   602→            <Button
   603→              key={p}
   604→              variant={p === page ? 'default' : 'outline'}
   605→              size="icon"
   606→              className="h-8 w-8"
   607→              onClick={() => goTo(p)}
   608→              aria-current={p === page ? 'page' : undefined}
   609→            >
   610→              {p}
   611→            </Button>
   612→          ),
   613→        )}
   614→        <Button
   615→          variant="outline"
   616→          size="sm"
   617→          disabled={page >= totalPages}
   618→          onClick={() => goTo(page + 1)}
   619→        >
   620→          Berikutnya

   656→/*  Main                                                                       */
   657→/* -------------------------------------------------------------------------- */
   658→
   659→export function ArticlesTable({
   660→  items,
   661→  total,
   662→  page,
   663→  pageSize,
   664→  q,
   665→  categoryFilter,
   666→  statusFilter,
   667→  sort,
   668→  categories,
   669→}: ArticlesTableProps) {
   670→  const router = useRouter()
   671→  const [selected, setSelected] = useState<Set<string>>(new Set())
   672→
   673→  // Reset selection bila filter berubah — pakai derived-state pattern
   674→  // (call setState during render with condition) supaya tidak trigger lint
   675→  // effect-cascade warning.
   676→  const filterKey = `${page}|${q}|${categoryFilter}|${statusFilter}|${sort}`
   677→  const [prevFilterKey, setPrevFilterKey] = useState(filterKey)
   678→  if (prevFilterKey !== filterKey) {
   679→    setPrevFilterKey(filterKey)
   680→    setSelected(new Set())
   681→  }
   682→
   683→  const allSelected = items.length > 0 && items.every((i) => selected.has(i.id))
   684→  const someSelected = items.some((i) => selected.has(i.id))
   685→
   686→  function toggleAll() {
   687→    setSelected((prev) => {
   688→      const next = new Set(prev)
   689→      if (allSelected) {
   690→        items.forEach((i) => next.delete(i.id))
   691→      } else {
   692→        items.forEach((i) => next.add(i.id))
   693→      }
   694→      return next
   695→    })
   696→  }
   697→  function toggleOne(id: string) {
   698→    setSelected((prev) => {
   699→      const next = new Set(prev)
   700→      if (next.has(id)) next.delete(id)
   701→      else next.add(id)
   702→      return next
   703→    })
   704→  }
   705→
   706→  return (
   707→    <div className="space-y-3">
   708→      <FilterBar
   709→        q={q}
   710→        categoryFilter={categoryFilter}
   711→        statusFilter={statusFilter}
   712→        sort={sort}
   713→        categories={categories}
   714→      />
   715→
   716→      <Card className="overflow-hidden p-0">
   717→        <Table>
   718→          <TableHeader>
   719→            <TableRow>
   720→              <TableHead className="w-[40px]">
   721→                <Checkbox
   722→                  checked={allSelected || (someSelected && 'indeterminate')}
   723→                  onCheckedChange={toggleAll}
   724→                  aria-label="Pilih semua artikel di halaman ini"
   725→                />
   726→              </TableHead>
   727→              <TableHead className="min-w-[280px]">Judul</TableHead>
   728→              <TableHead className="w-[140px]">Kategori</TableHead>
   729→              <TableHead className="w-[100px]">Status</TableHead>
   730→              <TableHead className="w-[90px] text-right">Views</TableHead>
   731→              <TableHead className="w-[140px]">Penulis</TableHead>
   732→              <TableHead className="w-[110px]">Tgl</TableHead>
   733→              <TableHead className="w-[60px] text-right">Aksi</TableHead>
   734→            </TableRow>
   735→          </TableHeader>
   736→          <TableBody>
   737→            {items.length === 0 ? (
   738→              <TableRow className="hover:bg-transparent">
   739→                <TableCell colSpan={8} className="p-0">
   740→                  <EmptyState q={q} />
   741→                </TableCell>
   742→              </TableRow>
   743→            ) : (
   744→              items.map((row) => (
   745→                <TableRow key={row.id} data-state={selected.has(row.id) ? 'selected' : undefined}>
   746→                  <TableCell>
   747→                    <Checkbox
   748→                      checked={selected.has(row.id)}
   749→                      onCheckedChange={() => toggleOne(row.id)}
   750→                      aria-label={`Pilih ${row.title}`}
   751→                    />
   752→                  </TableCell>
   753→                  <TableCell className="max-w-[420px]">
   754→                    <div className="flex flex-col gap-1">
   755→                      <Link
   756→                        href={`/admin/articles/${row.id}/edit`}
   757→                        className="line-clamp-2 font-medium text-foreground hover:text-amber-600 hover:underline"
   758→                      >
   759→                        {row.title}
   760→                      </Link>
   761→                      <div className="flex items-center gap-1.5">
   762→                        {row.isFeatured && (
   763→                          <Badge
   764→                            variant="outline"
   765→                            className="border-amber-300 bg-amber-50 text-amber-700 dark:bg-amber-900/30 dark:text-amber-200 dark:border-amber-700"
   766→                          >
   767→                            <Star className="h-3 w-3" />
   768→                            Featured
   769→                          </Badge>
   770→                        )}
   771→                        {row.isBreaking && (
   772→                          <Badge
   773→                            variant="outline"
   774→                            className="border-red-300 bg-red-50 text-red-700 dark:bg-red-900/30 dark:text-red-200 dark:border-red-700"
   775→                          >
   776→                            <Sparkles className="h-3 w-3" />
   777→                            Breaking
   778→                          </Badge>
   779→                        )}
   780→                        <span className="truncate text-[11px] text-muted-foreground">
   781→                          /{row.slug}
   782→                        </span>
   783→                      </div>
   784→                    </div>
   785→                  </TableCell>
   786→                  <TableCell>
   787→                    <Badge
   788→                      variant="outline"
   789→                      className={cn('whitespace-nowrap', categoryBadgeClass(row.category.color))}
   790→                    >
   791→                      {row.category.name}
   792→                    </Badge>
   793→                  </TableCell>
   794→                  <TableCell>
   795→                    <Badge
   796→                      variant="outline"
   797→                      className={cn('whitespace-nowrap', statusBadgeClass(row.status))}
   798→                    >
   799→                      {row.status}
   800→                    </Badge>
   801→                  </TableCell>
   802→                  <TableCell className="text-right tabular-nums">
   803→                    {fmtViews(row.viewCount)}
   804→                  </TableCell>
   805→                  <TableCell className="text-xs text-muted-foreground">
   806→                    {row.authorName}
   807→                  </TableCell>
   808→                  <TableCell className="whitespace-nowrap text-xs text-muted-foreground">
   809→                    {fmtDate(row.publishedAt ?? row.createdAt)}
   810→                  </TableCell>
   811→                  <TableCell className="text-right">
   812→                    <RowActions row={row} />
   813→                  </TableCell>
   814→                </TableRow>
   815→              ))
   816→            )}
   817→          </TableBody>
   818→        </Table>
   819→      </Card>
   820→
   821→      <BulkBar
   822→        selectedIds={selected}
   823→        onClear={() => setSelected(new Set())}
   824→        onDone={() => router.refresh()}
   825→      />
   826→
   827→      <PaginationBar total={total} page={page} pageSize={pageSize} />
   828→    </div>
   829→  )
   830→}
